Hazel and Fumola, Side by Side

Matthew, while the game's World rule was being written in Fumola for ^report(27) (^ticket(542)): would it be helpful to compare a document comparing the syntax of Hazel vs Fumola, to make translations easier next time? This is that document (^ticket(543)). Every Fumola line in it was run on Fumola's wasm build (667a34d); it grows as more of the rules are translated.

Traps, found translating#

  • Text != is inverted. "a" != "b" is false and "a" != "a" is true (run on the wasm build; the Neq arm for text in crates/fumola_semantics/src/lib/vm_ops.rs). Write not (a == b), or (a == b) == false.
  • No text patterns. case "E" never matches (patterns take Unit, Null and Nat literals only). Test text with == in an if.
  • Nat and Int apart. A Nat minus a larger Nat is an Int; Int + Nat is a type error; <=, >, >= between them are missing (only < mixes); Int(4) == Nat(4) is false, and they are different map keys. Keep coordinates Nat, and check an edge before stepping off it (the World rule's step).
  • No escapes in text. A literal's quotes are stripped, nothing more: no \" or \\ inside.
  • No Float comparison. Fumola's < has no Float case (found porting the Trap room, whose random wiring compares draws with 0.5). matthewhammer/kids-choice-games#4 passes the draws as Nats in millionths instead: exact at the threshold, but the rule's globals no longer match Hazel's shape.
  • A tuple payload, twice parenthesised. #GoSquare((c, r, back, #Done)): the outer pair is the application, the inner the tuple.

One rule, both ways: the World room's step#

The game's shipped Hazel rule (25-scripting.js, SCRIPT_WORLD, with Common's helpers it uses) and the Fumola one written for ^report(27). Both answer GoSquare(c, r, back, Done) or Done for the same moves.

Hazel
# from Common #
let member = fun (s, xs) : (String, [String]) -> case xs | [] => false | x :: rest => x == s || member(s, rest) end;
let dir = fun d : String -> if d == "N" then (0, 0 - 1) else if d == "NE" then (1, 0 - 1) else if d == "E" then (1, 0) else if d == "SE" then (1, 1) else if d == "S" then (0, 1) else if d == "SW" then (0 - 1, 1) else if d == "W" then (0 - 1, 0) else if d == "NW" then (0 - 1, 0 - 1) else (0, 0);
let opp = fun d : String -> if d == "N" then "S" else if d == "NE" then "SW" else if d == "E" then "W" else if d == "SE" then "NW" else if d == "S" then "N" else if d == "SW" then "NE" else if d == "W" then "E" else if d == "NW" then "SE" else ""
# the World room #
# Places with no way out on a side, whatever their ways say: the waterfall's N, its cliff. #
let no_way = fun (kind, d) : (String, String) -> kind == "waterfall" && d == "N";
let find = fun (cs, c, r) : ([Cell], Int, Int) -> case cs | [] => None | x :: rest => if x.col == c && x.row == r then Some(x) else find(rest, c, r) end;
# A way out walked through: into the square beside it, coming in from the side facing back, #
# if that square is on the map and has a way back; else nowhere. #
let on_event = fun (g, e) : (Globals, Event) ->
  case e
  | Way(d) =>
    let (dx, dy) = Common.dir(d) in
    let c = g.at_col + dx in
    let r = g.at_row + dy in
    let back = Common.opp(d) in
    if c < 0 || r < 0 || c >= g.cols || r >= g.rows then Done
    else (case find(g.cells, c, r)
      | Some(x) => if Common.member(back, x.ways) && !no_way(x.kind, back) then GoSquare(c, r, back, Done) else Done
      | None => Done
      end)
  | _ => Done
  end
Fumola
func no_way(kind, d) { kind == "waterfall" and d == "N" };
func member(s, xs) { var found = false; for (x in xs.vals()) { if (x == s) { found := true } }; found };
func opp(d) { if (d == "N") { "S" } else if (d == "NE") { "SW" } else if (d == "E") { "W" } else if (d == "SE") { "NW" } else if (d == "S") { "N" } else if (d == "SW") { "NE" } else if (d == "W") { "E" } else if (d == "NW") { "SE" } else { "" } };
func step(c, r, d, cols, rows) {
  let up = d == "N" or d == "NE" or d == "NW";
  let down = d == "S" or d == "SE" or d == "SW";
  let left = d == "W" or d == "NW" or d == "SW";
  let right = d == "E" or d == "NE" or d == "SE";
  if ((up and r == 0) or (left and c == 0)) { null }
  else {
    let c2 = if (left) { c - 1 } else if (right) { c + 1 } else { c };
    let r2 = if (up) { r - 1 } else if (down) { r + 1 } else { r };
    if (c2 < cols and r2 < rows) { ?(c2, r2) } else { null }
  }
};
func decide(x, back, c, r) { if (member(back, x.ways) and (no_way(x.kind, back) == false)) { #GoSquare((c, r, back, #Done)) } else { #Done } };
func find_at(cs, i, n, c, r) { if (i == n) { null } else { let x = cs[i]; if (x.col == c and x.row == r) { ?x } else { find_at(cs, i + 1, n, c, r) } } };
func world_list(g, e) {
  switch (e) {
    case (#Way(d)) {
      switch (step(g.at_col, g.at_row, d, g.cols, g.rows)) {
        case (?(c, r)) { switch (find_at(g.cells, 0, g.cells.size(), c, r)) { case (?x) { decide(x, opp(d), c, r) }; case null { #Done } } };
        case null { #Done }
      }
    };
    case _ { #Done }
  }
}
